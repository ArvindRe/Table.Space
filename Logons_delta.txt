SELECT s.snap_id,
        s.begin_interval_time,
        st.value - LAG(st.value) OVER (ORDER BY s.snap_id) AS logons_delta
    FROM    dba_hist_snapshot s
    JOIN    dba_hist_sysstat st
        ON  st.snap_id          = s.snap_id
        AND st.instance_number  = s.instance_number  -- ADD THIS
    WHERE   st.stat_name        = 'logons cumulative'
    AND     s.instance_number   = 1
    ORDER   BY s.snap_id DESC
    FETCH   FIRST 12 ROWS ONLY;

SELECT inst_id, count(*) sessions,
        SUM(CASE WHEN status='ACTIVE' THEN 1 ELSE 0 END) active,
        SUM(CASE WHEN status='INACTIVE' THEN 1 ELSE 0 END) inactive
FROM gv$session where type<>'BACKGROUND'
GROUP BY inst_id
ORDER BY inst_id;


SELECT inst_id,
        resource_name,
        current_utilization,
        max_utilization,
        limit_value
    FROM    gv$resource_limit
    WHERE   inst_id = 1
    AND     resource_name IN ('processes','sessions','transactions')
    ORDER   BY resource_name;

SELECT machine, program,
        COUNT(*)                    AS session_count,
        AVG(last_call_et)           AS avg_idle_secs,
        MIN(logon_time)             AS earliest_logon,
        MAX(logon_time)             AS latest_logon
    FROM    gv$session
    WHERE   inst_id = 1
    GROUP   BY machine, program
    ORDER   BY session_count DESC;
